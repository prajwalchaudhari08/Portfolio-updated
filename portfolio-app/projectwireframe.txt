        function init() {
            // Scene Setup
            scene = new THREE.Scene();
            scene.fog = new THREE.FogExp2(0x050510, 0.015); // Dark synthwave fog

            // Camera Setup
            camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
            
            // Renderer Setup
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); // Performance optimization
            document.getElementById('canvas-container').appendChild(renderer.domElement);

            // Lighting
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.2);
            scene.add(ambientLight);
            
            const dirLight = new THREE.DirectionalLight(0x7928ca, 1);
            dirLight.position.set(100, 200, 50);
            scene.add(dirLight);

            // Build Game Objects
            createEnvironment();
            createTrack();
            createCar();
            createCheckpoints();

            // Event Listeners
            window.addEventListener('resize', onWindowResize, false);
            document.addEventListener('keydown', onKeyDown);
            document.addEventListener('keyup', onKeyUp);

            // Start loop (paused until start button clicked)
            renderer.render(scene, camera);

        function createEnvironment() {
            // Neon Grid Floor
            const gridHelper = new THREE.GridHelper(1000, 100, 0x7928ca, 0x111122);
            gridHelper.position.y = -10;
            scene.add(gridHelper);

            // Distant Stars/Particles
            const particleCount = 1000;
            const geo = new THREE.BufferGeometry();
            const pos = new Float32Array(particleCount * 3);
            for(let i=0; i<particleCount*3; i++) {
                pos[i] = (Math.random() - 0.5) * 800;
            }
            geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
            const mat = new THREE.PointsMaterial({color: 0x00bfff, size: 0.5, transparent: true, opacity: 0.6});
            const particles = new THREE.Points(geo, mat);
            scene.add(particles);
        }

        function createTrack() {
            // Define a winding closed loop for the track
            const points = [
                new THREE.Vector3(0, 0, 0),
                new THREE.Vector3(0, 0, -100),
                new THREE.Vector3(-100, 0, -200),
                new THREE.Vector3(-200, 0, -150),
                new THREE.Vector3(-250, 10, -50), // Slight elevation
                new THREE.Vector3(-150, 0, 100),
                new THREE.Vector3(50, 0, 200),
                new THREE.Vector3(150, -10, 100), // Dip
                new THREE.Vector3(100, 0, 0),
                new THREE.Vector3(0, 0, 0)
            ];

            trackCurve = new THREE.CatmullRomCurve3(points);
            trackCurve.closed = true;

            // Track Geometry - Extrude a flat rectangle along the curve
            const trackWidth = 8;
            const shape = new THREE.Shape();
            shape.moveTo(-trackWidth/2, -0.2);
            shape.lineTo(trackWidth/2, -0.2);
            shape.lineTo(trackWidth/2, 0.2);
            shape.lineTo(-trackWidth/2, 0.2);
            shape.lineTo(-trackWidth/2, -0.2);

            const extrudeSettings = {
                steps: 300,
                extrudePath: trackCurve
            };

            const trackGeometry = new THREE.ExtrudeGeometry(shape, extrudeSettings);
            
            // Dark asphalt material
            const trackMaterial = new THREE.MeshStandardMaterial({ 
                color: 0x111111,
                roughness: 0.8,
                metalness: 0.2
            });
            const trackMesh = new THREE.Mesh(trackGeometry, trackMaterial);
            scene.add(trackMesh);

            // Neon Edge Lines
            const edges = new THREE.EdgesGeometry(trackGeometry, 20);
            const lineMat = new THREE.LineBasicMaterial({ color: 0x00bfff, linewidth: 2 });
            const neonEdges = new THREE.LineSegments(edges, lineMat);
            scene.add(neonEdges);
        }

        function createCar() {
            carGroup = new THREE.Group();

            // Materials
            const bodyMat = new THREE.MeshStandardMaterial({ color: 0x222222, metalness: 0.8, roughness: 0.2 });
            const accentMat = new THREE.MeshStandardMaterial({ color: 0xff007f, emissive: 0x330011 });
            const tireMat = new THREE.MeshStandardMaterial({ color: 0x050505, roughness: 0.9 });

            // Main Chassis
            const chassisGeo = new THREE.BoxGeometry(1.5, 0.5, 4);
            const chassis = new THREE.Mesh(chassisGeo, bodyMat);
            chassis.position.y = 0.5;
            carGroup.add(chassis);

            // Nose
            const noseGeo = new THREE.ConeGeometry(0.75, 2, 4);
            const nose = new THREE.Mesh(noseGeo, bodyMat);
            nose.rotation.x = Math.PI / 2;
            nose.rotation.y = Math.PI / 4; // rotate square base to look like diamond wedge
            nose.position.set(0, 0.5, 3);
            carGroup.add(nose);

            // Cockpit Dome
            const domeGeo = new THREE.SphereGeometry(0.6, 16, 16, 0, Math.PI * 2, 0, Math.PI/2);
            const domeMat = new THREE.MeshStandardMaterial({ color: 0x000000, metalness: 1, roughness: 0 });
            const dome = new THREE.Mesh(domeGeo, domeMat);
            dome.position.set(0, 0.75, -0.5);
            dome.scale.z = 1.5;
            carGroup.add(dome);

            // Front Wing
            const fWingGeo = new THREE.BoxGeometry(3, 0.1, 0.8);
            const fWing = new THREE.Mesh(fWingGeo, accentMat);
            fWing.position.set(0, 0.2, 3.8);
            carGroup.add(fWing);

            // Rear Wing
            const rWingGeo = new THREE.BoxGeometry(2.5, 0.1, 1);
            const rWing = new THREE.Mesh(rWingGeo, accentMat);
            rWing.position.set(0, 1.2, -1.8);
            carGroup.add(rWing);
            
            // Rear Wing Pillars
            const pillarGeo = new THREE.BoxGeometry(0.1, 0.8, 0.5);
            const p1 = new THREE.Mesh(pillarGeo, bodyMat);
            p1.position.set(-0.8, 0.8, -1.8);
            const p2 = p1.clone();
            p2.position.set(0.8, 0.8, -1.8);
            carGroup.add(p1, p2);

            // Wheels
            const wheelGeo = new THREE.CylinderGeometry(0.5, 0.5, 0.6, 16);
            wheelGeo.rotateZ(Math.PI / 2);

        function createCheckpoints() {
            const spacing = 1.0 / projectsData.length;
            
            projectsData.forEach((project, index) => {
                const progress = (index * spacing) + 0.05; // Offset slightly from start
                const point = trackCurve.getPointAt(progress);
                const tangent = trackCurve.getTangentAt(progress);
                
                // Arch geometry
                const archGroup = new THREE.Group();
                archGroup.position.copy(point);
                
                // Orient arch perpendicular to track
                const up = new THREE.Vector3(0, 1, 0);
                const right = new THREE.Vector3().crossVectors(tangent, up).normalize();
                
                // Construct glowing ring
                const ringGeo = new THREE.TorusGeometry(8, 0.3, 16, 50);
                // Rotate to stand up
                ringGeo.rotateX(Math.PI / 2);
                ringGeo.rotateY(Math.PI / 2);
                
                const ringMat = new THREE.MeshStandardMaterial({ 
                    color: project.color, 
                    emissive: project.color,
                    emissiveIntensity: 0.5,
                    transparent: true,
                    opacity: 0.8
                });
                
                const ring = new THREE.Mesh(ringGeo, ringMat);
                
                // Orient ring to face down the track
                const lookAtPos = point.clone().add(tangent);
                archGroup.lookAt(lookAtPos);
                
                archGroup.add(ring);

                // Add a small ground marker
                const markerGeo = new THREE.PlaneGeometry(10, 4);
                const markerMat = new THREE.MeshBasicMaterial({ 
                    color: project.color, 
                    transparent: true, 
                    opacity: 0.4,
                    side: THREE.DoubleSide
                });
                const marker = new THREE.Mesh(markerGeo, markerMat);
                marker.rotation.x = -Math.PI / 2;
                marker.position.y = 0.5; // Slightly above track
                archGroup.add(marker);

                scene.add(archGroup);

                // Store checkpoint data for collision detection
                checkpoints.push({
                    progress: progress,
                    data: project,
                    ring: ring
                });
            });
        }

        function updateCarPosition(progress) {
            // Get position and orientation from curve
            const position = trackCurve.getPointAt(progress);
            const tangent = trackCurve.getTangentAt(progress);
            
            // Update Car Position
            carGroup.position.copy(position);
            
            // Update Car Rotation to look along the curve
            const lookAtTarget = position.clone().add(tangent);
            carGroup.lookAt(lookAtTarget);

            // Add slight roll based on curvature (optional, makes it look more dynamic)
            // carGroup.rotation.z += some_calculated_roll; 
        }

        function animate() {
            if (!gameStarted) return;
            requestAnimationFrame(animate);

            // 1. Calculate Car Physics
            if (keys.w) {
                carAcceleration = accelerationRate;
            } else if (keys.s) {
                carAcceleration = -brakingRate;
            } else {
                carAcceleration = 0;
            }

            carVelocity += carAcceleration;
            carVelocity *= friction; // apply friction
            
            // Cap speed
            if (carVelocity > maxSpeed) carVelocity = maxSpeed;
            if (carVelocity < -maxSpeed/2) carVelocity = -maxSpeed/2; // slower reverse

            // Update progress
            carProgress += carVelocity;
            
            // Loop track
            if (carProgress > 1) carProgress -= 1;
            if (carProgress < 0) carProgress += 1;

            // Update 3D car
            updateCarPosition(carProgress);

            // 2. Camera Follow Logic
            // Smoothly move camera towards the target position attached behind the car
            const targetPos = new THREE.Vector3();
            cameraTarget.getWorldPosition(targetPos);
            
            // Lerp camera position
            camera.position.lerp(targetPos, 0.1);
            
            // Look slightly ahead of the car
            const lookAheadProgress = (carProgress + 0.02) % 1;
            const lookAheadPoint = trackCurve.getPointAt(lookAheadProgress);
            // Smooth lookat
            const currentLookAt = new THREE.Vector3(0,0,-1).applyQuaternion(camera.quaternion).add(camera.position);
            currentLookAt.lerp(lookAheadPoint, 0.1);
            camera.lookAt(currentLookAt);

            // 3. UI Updates
            // Speedometer
            const displaySpeed = Math.abs(Math.round((carVelocity / maxSpeed) * 320)); // arbitrary km/h scaling
            speedometer.textContent = `SPD: ${displaySpeed.toString().padStart(3, '0')} km/h`;

            // Minimap Dot
            // Map carProgress (0-1) to an angle around the circular minimap
            const angle = carProgress * Math.PI * 2 - (Math.PI / 2);
            const radius = 60; // radius of minimap interior
            const mmX = Math.cos(angle) * radius + 75; // 75 is center
            const mmY = Math.sin(angle) * radius + 75;
            minimapDot.style.left = `${mmX}px`;
            minimapDot.style.top = `${mmY}px`;

            // 4. Checkpoint Detection
            let nearestCheckpoint = null;
            let minDistance = 0.02; // Threshold to trigger UI (approx length of car)

            checkpoints.forEach((cp, index) => {
                // Handle looping edge case for distance calculation
                let dist = Math.abs(carProgress - cp.progress);
                if (dist > 0.5) dist = 1.0 - dist;

                // Make ring pulse
                cp.ring.scale.setScalar(1 + Math.sin(Date.now() * 0.005 + index) * 0.05);

                if (dist < minDistance) {
                    nearestCheckpoint = { index: index, data: cp.data };
                }
            });

            // Show UI if near a checkpoint AND moving slowly enough to read
            if (nearestCheckpoint && Math.abs(carVelocity) < (maxSpeed * 0.4)) {
                if (currentActiveCheckpoint !== nearestCheckpoint.index) {
                    currentActiveCheckpoint = nearestCheckpoint.index;
                    updateUI(nearestCheckpoint.data);
                }
            } else {
                if (currentActiveCheckpoint !== -1) {
                    currentActiveCheckpoint = -1;
                    updateUI(null);
                }
            }

            // Render Scene
            renderer.render(scene, camera);
        }
